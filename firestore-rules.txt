rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /bets/{betId} {
      allow read, write: if true;
    }
    match /games/{gameId} {
      allow read, write: if true;
    }
    match /live/{docId} {
      allow read: if true;
      allow write: if true; // O Worker escreve aqui
    }
  }
}
